cmake_minimum_required(VERSION 3.25)
project(PlasmaTrayMotion VERSION 0.1.9 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PROJECT_DEP_VERSION 6.7.0)
set(BUILD_TESTING OFF CACHE BOOL "Build upstream tests")
find_package(ECM 6.26 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(ECMQtDeclareLoggingCategory)
include(ECMQmlModule)
find_package(Qt6 6.11 CONFIG REQUIRED COMPONENTS Core DBus Gui Quick Widgets)
find_package(Qt6GuiPrivate CONFIG REQUIRED)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.12)
    message(FATAL_ERROR "The native drag image adapter requires review for Qt ${Qt6_VERSION}; supported Qt branch: 6.11.")
endif()
find_package(KF6 6.26 REQUIRED COMPONENTS Config CoreAddons I18n IconThemes KIO JobWidgets Service ItemModels WindowSystem XmlGui StatusNotifierItem)
find_package(Plasma 6.7 CONFIG REQUIRED)
find_package(PlasmaQuick 6.7 CONFIG REQUIRED)
set(plasma-workspace_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/work/source")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config-X11.h" "#define HAVE_X11 1\n")
include_directories("${CMAKE_CURRENT_BINARY_DIR}")
add_subdirectory("${plasma-workspace_SOURCE_DIR}/libdbusmenuqt" dbusmenu EXCLUDE_FROM_ALL)
target_include_directories(dbusmenuqt PUBLIC "${plasma-workspace_SOURCE_DIR}/libdbusmenuqt")
add_subdirectory("${plasma-workspace_SOURCE_DIR}/applets/systemtray" systemtray)
